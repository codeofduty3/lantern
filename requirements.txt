# --- Intel macOS (x86_64) only ---------------------------------------------
# The newest PyTorch wheel for Intel Macs is 2.2.2: its C extension is built against
# NumPy 1.x (numpy 2 disables the torch<->numpy bridge) and it has no
# torch.serialization.safe_globals (added in 2.6), so opencv 5 (numpy>=2) is held back
# on this platform. transformers is pinned exactly in the list above, which already
# satisfies its old `transformers<5` constraint here.
numpy<2; sys_platform == "darwin" and platform_machine == "x86_64"
opencv-python<5; sys_platform == "darwin" and platform_machine == "x86_64"
opencv-python-headless<5; sys_platform == "darwin" and platform_machine == "x86_64"
